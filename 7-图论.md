## 图论 ##

#### (`有向图`)强联通分量缩点 ####

强连通分量缩点后的图称为 SCC。以 O(N + M) 的复杂度完成上述全部操作。

缩点后的图拥有拓扑序

```cpp
#include <bits/stdc++.h>
#define endl '\n'

using ll = long long;

constexpr int N = 2e5 + 10;
constexpr int mod = 998244353;

using namespace std;

struct SCC {
    int n, now, cnt;
    vector<vector<int>> ver;
    vector<int> dfn, low, col, id, S;

    SCC(int n) : n(n), ver(n + 1), low(n + 1), id(n + 1) {
        dfn.resize(n + 1, - 1);
        col.resize(n + 1, - 1);
        now = cnt = 0;
    }

    void add(int x, int y) {
        ver[x].push_back(y);
    }

    void tarjan(int x) {
        dfn[x] = low[x] = now++;
        S.push_back(x);
        for (auto y : ver[x]) {
            if (dfn[y] == -1) {
                tarjan(y);
                low[x] = min(low[x], low[y]);
            } else if (col[y] == -1) {
                low[x] = min(low[x], dfn[y]);
            }
        }
        if (dfn[x] == low[x]) {
            int pre;
            cnt++;
            id[cnt] = x;
            do {
                pre = S.back();
                col[pre] = cnt;
                S.pop_back();
            } while (pre != x);
        }
    }

    auto work () {
        for (int i = 1; i <= n; ++i) {
            if (dfn[i] == -1) tarjan(i);
        }

        vector<int> siz(cnt + 1);
        vector<vector<int>> adj(cnt + 1);

        for (int i = 1; i <= n; ++i) {
            siz[col[i]]++;
            for (auto j : ver[i]) {
                int x = col[i], y = col[j];
                if (x != y) {
                    adj[x].push_back(y);
                }
            }
        }

        //return tuple{cnt, adj, col, sz};
        return tuple{cnt, col, siz, id};
    }
};

void solve() {
    int n, m;
    cin >> n >> m;
    SCC scc(n);
    vector<pair<int, int>> e(n + 1);
    for (int i = 1; i <= m; ++i) {
        int u, v;
        cin >> u >> v;
        e[i] = {u, v};
        scc.add(u, v);
    }

    auto [cnt, col, sz, id] = scc.work();

    vector<int> in(n + 1);
    for (int i = 1; i <= m; ++i) {
        auto [u, v] = e[i];
        if (col[u] != col[v]) {
            in[col[v]]++;
        }
    }
    vector<int> ans;
    // for (int i = 1; i <= cnt; ++i){
    //     cout << id[i] << " \n"[i == cnt];
    // }
    for (int i = 1; i <= cnt; ++i) {
        if (in[i] == 0) ans.push_back(id[i]);
    }
    sort(ans.begin(), ans.end());
    cout << ans.size() << endl;
    for (auto x : ans) {
        cout << x << " ";
    }
    cout << endl;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //cin >> t;
    while (t--) solve();
    return 0;
}
```

#### ('无向图')割点缩点 ####

割点缩点后的图称为点双连通图 (V - DCC)，该模板可以在 O(N + M) 复杂度内求解图中全部割点、划分点双（颜色相同的点位于同一个点双连通分量中）。

```cpp
VDCC(int n) : n(n) {
    ver.resize(n + 1);
    dfn.resize(n + 1);
    low.resize(n + 1);
    col.resize(2 * n + 1);
    point.resize(n + 1);
    S.clear();
    cnt = now = 0;
}

void add(int u, int v) {
    if (u == v) return;
    ver[u].push_back(v);
    ver[v].push_back(u);
}


void tarjan(int u, int rt) {
    low[u] = dfn[u] = ++now;
    S.push_back(u);
    if (u == rt && !ver[u].size()) {
        ++cnt;
        col[cnt].push_back(u);
        return;
    }
    int flag = 0;
    for (auto v : ver[u]) {
        if (!dfn[v]) {
            tarjan(v, rt);
            low[u] = min(low[u], low[v]);
            if (dfn[u] <= low[v]) {
                flag++;
                if (u != rt || flag > 1) {
                    point[u] = 1;
                }
                int pre = 0;
                cnt++;
                do {
                    pre = S.back();
                    col[cnt].push_back(pre);
                    S.pop_back();
                } while (pre != v);
                col[cnt].push_back(u);
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
}
pair<int, vector<vector<int>>> rebuild() { // 新图的顶点数量，新图
    work();
    vector<vector<int>> adj(cnt + 1);
    for (int i = 1; i <= cnt; ++i) {
        if (!col[i].size()) {
            continue;
        }
        for (auto j : col[i]) {
            if (point[j]) {
                adj[i].push_back(point[j]);
                adj[point[j]].push_back(i);
            }
        }
    }
    return {cnt, adj};
}

void work() {
    for (int i = 1; i <= n; ++i) {
        if (!dfn[i]) tarjan(i, i);
    }
}
};
```

#### ('无向图') 边双，不求桥 ####
 
割边缩点后的图称为边双连通图 (E - DCC)，该模板可以在 O(N + M) 复杂度内求解图中全部割
边、划分边双（颜色相同的点位于同一个边双连通分量中）。

性质补充：对于一个边双，删去任意边后依旧联通；对于边双中的任意两点，一定存在两条不相
交的路径连接这两个点（路径上可以有公共点，但是没有公共边）。

```cpp
struct EDCC {
    int n, now, cnt;
    vector<vector<pair<int, int>>> ver;
    vector<int> col, dfn, low, vis, ind, S;

    EDCC(int n) : n(n), ver(n + 1), col(n + 1), dfn(n + 1), low(n + 1), vis(n + 1), ind(n + 1) {
        now = cnt = 0;
    }

    void add(int u, int v, int id) {
        if (u == v) return;
        ver[u].emplace_back(v, id << 1);
        ver[v].emplace_back(u, id << 1 | 1);
    }

    void tarjan(int u, int fa) {
        low[u] = dfn[u] = ++now;
        S.push_back(u);
        vis[u] = 1;
        for (auto [v, t] : ver[u]) {
            if (t == (fa ^ 1)) continue;
            if (!dfn[v]) {
                tarjan(v, t);
                low[u] = min(low[u], low[v]);
            } else if (vis[v]) {
                low[u] = min(low[u], dfn[v]);
            }
        }
        if (low[u] == dfn[u]) {
            int pre;
            cnt++;
            do {
                pre = S.back();
                vis[pre] = 0;
                col[pre] = cnt;
                S.pop_back();
            } while (pre != u);
        }
    }

    auto work() {
        for (int i = 1; i <= n; ++i) if (!dfn[i]) tarjan(i, 0);
        vector<int> sz(cnt + 1);
        vector<vector<int>> adj(cnt + 1);
        for (int i = 1; i <= n; ++i) {
            int x = col[i];
            sz[x]++;
            for (auto [j, t] : ver[i]) {
                int y = col[j];
                if (x != y) {
                    adj[x].push_back(y);
                    ind[y]++;
                }
            }
        }

        return tuple{cnt, adj, col, sz, ind};
    }
};

```

#### 求桥 ####

下面代码实现了求割边，其中，当 isbridge[x] 为真时，(father[x],x) 为一条割边。
(`下面代码无法处理重边`)

```cpp
int low[MAXN], dfn[MAXN], dfs_clock;
bool isbridge[MAXN];
vector<int> G[MAXN];
int cnt_bridge;
int father[MAXN];

void tarjan(int u, int fa) {
    father[u] = fa;
    low[u] = dfn[u] = ++dfs_clock;
    for (int i = 0; i < G[u].size(); i++) {
        int v = G[u][i];
        if (!dfn[v]) {
            tarjan(v, u);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) {
                isbridge[v] = true;
                ++cnt_bridge;
            }
        } else if (dfn[v] < dfn[u] && v != fa) {
            low[u] = min(low[u], dfn[v]);
        }
    }
}
```

#### 求无向图中不在奇圈上的点 ####

1.如果一个双连通分量内的某些顶点在一个奇圈中（即双连通分量含有奇圈），那么这个双连通分量的其他顶点也在某个奇圈中；
2.如果一个双连通分量含有奇圈，则他必定不是一个二分图。反过来也成立，这是一个充要条件。
基于以上俩点， Tarjan求出每个点双连通图分量后，用交叉染色法判断是否存在奇圈！

```cpp
#include<bits/stdc++.h>
using namespace std;
const int maxn = 2e6 + 50;
const int N = 1050;
struct node{
    int to, next;
} edge[maxn];
int top, stack1[maxn];
int tot, cnt, dccn;
int dfn[N], low[N];
int head[N];
vector<int> dcc[N];
void add(int from, int to){
    edge[++cnt].to = to;edge[cnt].next = head[from];head[from] = cnt;
}
void ins(int u, int v){add(u, v); add(v, u);}
void Tarjan(int u, int rt){
    dfn[u] = low[u] = ++tot;
    if(rt == u && head[u] == -1){
        dccn++;
        dcc[dccn].clear();
        dcc[dccn].push_back(u);
        return;
    }
    stack1[++top] = u;
    for(int i = head[u];i != -1;i = edge[i].next){
        int v = edge[i].to;
        if(!dfn[v]){
            Tarjan(v, rt);
            low[u] = min(low[u], low[v]);
            if(low[v] >= dfn[u]){
                dccn++;
                dcc[dccn].clear();
                dcc[dccn].push_back(u);
                while (1){
                    int w = stack1[top];
                    dcc[dccn].push_back(w);
                    top--;
                    if (w == v){ // 做到子树全部弹出为止，不是u，不然v的兄弟也会被弹出
                        break;
                    }
                }
            }
        }
        else low[u] = min(low[u], dfn[v]);
    }
}
int now[N], col[N], vis[N];
bool dfs(int u, int c){
    col[u] = c;
    for(int i = head[u];i != -1;i = edge[i].next){
        int v = edge[i].to;
        if(!now[v]) continue;
        if(col[v] == c) return false;
        if(col[v] == 0 && !dfs(v, 3 - c)) return false;
    }
    return true;
}
int mp[N][N];
void init(){
    cnt = 0; tot = 0, dccn = 0, top = 0;
    memset(head, -1, sizeof(head));
    memset(mp, 0, sizeof(mp));
    memset(low, 0, sizeof(low));
    memset(dfn, 0, sizeof(dfn));
    memset(vis, 0, sizeof(vis));
}
int main()
{
    std::ios::sync_with_stdio(false);
    int n, m;
    while(cin >> n >> m){
        if(n == 0 && m == 0) break;
        init();
        for(int i = 0;i < m;i++){
            int u, v;
            cin >> u >> v;
            mp[u][v] = mp[v][u] = 1;
        }
        for(int i = 1;i <= n;i++){
            for(int j = i + 1;j <= n;j++){
                if(mp[i][j] == 0){
                    ins(i, j);
                }
            }
        }
        for(int i = 1;i <= n;i++) if(!dfn[i]) Tarjan(i, i);
        for(int i = 1;i <= dccn;i++){
            memset(now, 0, sizeof(now));
            memset(col, 0, sizeof(col));
            for(int j = 0;j < dcc[i].size();j++){
                now[dcc[i][j]] = 1;
            }
            if(!dfs(dcc[i][0], 1)){
                for(int j = 0;j < dcc[i].size();j++){
                    vis[dcc[i][j]] = 1;
                }
            }
        }
        int ans = 0;
        for(int i = 1;i <= n;i++) if(!vis[i]) ans++;
        cout << ans << endl;
    }
    return 0;
}
```

#### DAG最小路径覆盖 ####

```cpp
#include <bits/stdc++.h>
#define endl '\n'

using ll = long long;

constexpr int N = 2e5 + 10;
constexpr int mod = 998244353;

using namespace std;

struct SCC {
    int n, now, cnt;
    vector<vector<int>> ver;
    vector<int> dfn, low, col, id, S;

    SCC(int n) : n(n), ver(n + 1), low(n + 1), id(n + 1) {
        dfn.resize(n + 1, - 1);
        col.resize(n + 1, - 1);
        now = cnt = 0;
    }

    void add(int x, int y) {
        ver[x].push_back(y);
    }

    void tarjan(int x) {
        dfn[x] = low[x] = now++;
        S.push_back(x);
        for (auto y : ver[x]) {
            if (dfn[y] == -1) {
                tarjan(y);
                low[x] = min(low[x], low[y]);
            } else if (col[y] == -1) {
                low[x] = min(low[x], dfn[y]);
            }
        }
        if (dfn[x] == low[x]) {
            int pre;
            cnt++;
            id[cnt] = x;
            do {
                pre = S.back();
                col[pre] = cnt;
                S.pop_back();
            } while (pre != x);
        }
    }
};

void solve(){
    int n;
    cin >> n;
    vector<string> s(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> s[i];
    }
    SCC scc(n);
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (s[i].back() == s[j].front()) {
                scc.add(i, j);
            }
        }
    }

    for (int i = 1; i <= n; ++i) {
        if (scc.dfn[i] == -1) scc.tarjan(i);
    }

    int cnt = scc.cnt;

    vector<int> siz(cnt + 1);
    vector<vector<int>> adj(cnt + 1, vector<int>(cnt + 1));

    for (int i = 1; i <= n; ++i) {
        siz[scc.col[i]]++;
        for (auto j : scc.ver[i]) {
            int x = scc.col[i], y = scc.col[j];
            if (x != y) {
                adj[x][y] = 1;
            }
        }
    }

    for (int k = 1; k <= cnt; ++k) {
        for (int i = 1; i <= cnt; ++i) {
            for (int j = 1; j <= cnt; ++j) {
                if (i == j) continue;
                if (adj[i][k] && adj[k][j]) adj[i][j] = 1;
            }
        }
    }

    vector<int> vis(cnt + 1), pr(cnt + 1, -1);
    int tot = 0;
    auto dfs = [&](auto self, int u) -> bool {
        vis[u] = tot;
        for (int v = 1; v <= cnt; ++v) {
            if (adj[u][v] && (pr[v] == -1 || (vis[pr[v]] != tot && self(self, pr[v])))) {
                pr[v] = u;  
                return true;
            } 
        }
        return false;
    };
    int ans = 0;
    for (int i = 1; i <= cnt; ++i) {
        tot++;
        if (dfs(dfs, i)) ans++;
    }
    cout << cnt - ans << endl;
}

signed main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //cin >> t;
    while(t--) solve();
    return 0;
}
```
