```cpp
        ls[rt] = ls[old], rs[rt] = rs[old], sum[rt] = sum[old] + k;
        if (l == r) return;

        if (l == r) return sum[rt];
        int mid = l + r >> 1;
        if (p <= mid) return query(ls[rt], l, mid, p) + sum[rs[rt]];
        else return query(rs[rt], mid + 1, r, p);
    }
}T;

        int l, r;

    }
```

```cpp
#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long double db;
typedef long long ll;

const ll N = 6e5 + 10;
const ll mod = 998244353;
const ll inf32 = 0x3f3f3f3f;
const ll inf64 = 5e18;

int a[N], s[N];
struct Trie
{
    int cnt, rt[N], ch[N * 33][2], val[N * 33];
    void insert(int o, int lst, int v)
    {
        for (int i = 28; i >= 0; i--)
        {
            val[o] = val[lst] + 1; // 在原版本的基础上更新
            if ((v & (1 << i)) == 0)
            {
                if (!ch[o][0])
                    ch[o][0] = ++cnt;
                ch[o][1] = ch[lst][1];
                o = ch[o][0];
                lst = ch[lst][0];
            }
            else
            {
                if (!ch[o][1])
                    ch[o][1] = ++cnt;
                ch[o][0] = ch[lst][0];
                o = ch[o][1];
                lst = ch[lst][1];
            }
        }
        val[o] = val[lst] + 1;
    }

    int query(int o1, int o2, int v)
    {
        int ret = 0;
        for (int i = 28; i >= 0; i--)
        {
            int t = ((v & (1 << i)) ? 1 : 0);
            if (val[ch[o1][!t]] - val[ch[o2][!t]])
                ret += (1 << i), o1 = ch[o1][!t],
                                 o2 = ch[o2][!t]; // 尽量向不同的地方跳
            else
                o1 = ch[o1][t], o2 = ch[o2][t];
        }
        return ret;
    }
} st;

void solve()
{
    int n, q;
    cin >> n >> q;
    for (int i = 1; i <= n; ++i) cin >> a[i], s[i] = s[i - 1] ^ a[i];
    for (int i = 1; i <= n; ++i){
        st.rt[i] = ++st.cnt;
        st.insert(st.rt[i], st.rt[i - 1], s[i]);
    }
    while (q--){
        char op;
        cin >> op;
        if (op == 'A'){
            ++n;
            cin >> a[n];
            s[n] = s[n - 1] ^ a[n];
            st.rt[n] = ++st.cnt;
            st.insert(st.rt[n], st.rt[n - 1], s[n]);
        }else{
            int l, r, x;
            cin >> l >> r >> x;
            l--, r--;
            if (l == 0) cout << max(s[n] ^ x, st.query(st.rt[r], st.rt[0], s[n] ^ x)) << endl;
            else cout << st.query(st.rt[r], st.rt[l - 1], s[n] ^ x) << endl; 
        }
    }
}

signed main()
{
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int t = 1;
    // cin >> t;
    while (t--)
        solve();
    return 0;
}
```

```cpp
#include <bits/stdc++.h>
#define endl '\n'
#define pll pair<i64, i64>
#define tll tuple<i64, i64, i64>
#define all(a) a.begin() + 1, a.end()
using namespace std;
using db = long double;
using i64 = long long;
const i64 N = 3e5 + 10;
const i64 mod = 998244353;
const i64 inf32 = 1e9;
const i64 inf64 = 5e18;

int rt[N];
struct PersistentTree{
    int ls[N << 5], rs[N << 5], mn[N << 5], sum[N << 5], tot = 0;
    inline void build(int & rt, int l, int r){
        rt = ++tot;
        if (l == r) return;
        int mid = l + r >> 1;
        build(ls[rt], l, mid);
        build(rs[rt], mid + 1, r);
    }
    inline void update(int & rt, int old, int l, int r, int p, int k){
        rt = ++tot;
        ls[rt] = ls[old], rs[rt] = rs[old], mn[rt] = mn[old], sum[rt] = sum[old];
        if (l == r){
            mn[rt] = k;
            sum[rt]++;
            return;
        }
        int mid = l + r >> 1;
        if (p <= mid) update(ls[rt], ls[old], l, mid, p, k);
        else update(rs[rt], rs[old], mid + 1, r, p, k);
        mn[rt] = min(mn[ls[rt]], mn[rs[rt]]);
        sum[rt] = sum[ls[rt]] + sum[rs[rt]];
    }
    inline int query(int rt, int old, int p){
        if (mn[rt] >= p) return sum[rt] - sum[old];
        if (ls[rt] == 0) return 0;
        int res = 0;
        if (mn[ls[rt]] >= p) res = sum[ls[rt]] - sum[ls[old]] + query(rs[rt], rs[old], p);
        else res = query(ls[rt], ls[old], p);
        return res;
    }
}T;

void solve(){
    int n, q;
    cin >> n;
    T.build(rt[0], 1, n);
    vector<int> a(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        if (a[i] <= n) T.update(rt[i], rt[i - 1], 1, n, a[i], i);
        else rt[i] = rt[i - 1];
    }
    cin >> q;
    while (q--){
        int l, r;
        cin >> l >> r;
        cout << (r - l + 1) - T.query(rt[r], rt[l - 1], l) << endl;
    }
}

int main(){
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int t = 1;
    //cin >> t;
    while(t--) solve();
    return 0;
}
```

#### 可持久化并查集 ####

```cpp
#include <bits/stdc++.h>
#define endl '\n'
#define pll pair<i64, i64>
#define tll tuple<i64, i64, i64>
#define all(a) a.begin() + 1, a.end()
using namespace std;
using db = long double;
using i64 = long long;
const i64 N = 2e5 + 10;
const i64 mod = 998244353;
const i64 inf32 = 1e9;
const i64 inf64 = 5e18;

int n, m;
int rootfa[N], rootdep[N];
struct PersistentDSU{
    int ls[N << 5], rs[N << 5], val[N << 5], tot = 0, idx = 0;
    inline void build(int & rt, int l, int r){
        rt = ++tot;
        if (l == r){
            val[rt] = ++idx;
            return;
        }
        int mid = (l + r) >> 1;
        build(ls[rt], l, mid);
        build(rs[rt], mid + 1, r);
    }
    inline void update(int & rt, int old, int l, int r, int p, int k){
        rt = ++tot;
        ls[rt] = ls[old], rs[rt] = rs[old], val[rt] = val[old];
        if (l == r){
            val[rt] = k;
            return;
        }
        int mid = (l + r) >> 1;
        if (p <= mid) update(ls[rt], ls[old], l, mid, p, k);
        else update(rs[rt], rs[old], mid + 1, r, p, k);
    }
    inline int query(int rt, int l, int r, int p){
        if (l == r) return val[rt];
        int mid = (l + r) >> 1;
        if (p <= mid) return query(ls[rt], l, mid, p);
        else return query(rs[rt], mid + 1, r, p);
    }
    //找到第ver个版本中x的父亲
    inline int find(int ver, int x){
        //去ver版本线段树rootfa[ver]中查询x上的值
        int fx = query(rootfa[ver], 1, n, x);
        if (x != fx) x = find(ver, fx); //不能进行路径压缩
        return x;
    }
    //在ver个版本中合并集合x和集合y
    inline void merge(int ver, int x, int y){ 
        //千万要注意rootdep和rootfa!!!
        x = find(ver - 1, x);
        y = find(ver - 1, y);
        if (x == y){
            rootfa[ver] = rootfa[ver - 1];
            rootdep[ver] = rootdep[ver - 1];    
        }else{
            int depx = query(rootdep[ver - 1], 1, n, x);
            int depy = query(rootdep[ver - 1], 1, n, y);
            if (depx < depy){
                //fa[x] = y
                update(rootfa[ver], rootfa[ver - 1], 1, n, x, y);
                rootdep[ver] = rootdep[ver - 1];
            }else if (depy < depx){
                //fa[y] = x
                update(rootfa[ver], rootfa[ver - 1], 1, n, y, x);
                rootdep[ver] = rootdep[ver - 1];
            }else{
                //fa[x] = y
                update(rootfa[ver], rootfa[ver - 1], 1, n, x, y);
                update(rootdep[ver], rootdep[ver - 1], 1, n, y, depy + 1);
            }
        }
    }
}D;

void solve(){
    cin >> n >> m;
    D.build(rootfa[0], 1, n);
    for (int i = 1; i <= m; ++i){
        int op, a, b;
        cin >> op;
        if (op == 1){ //合并
            cin >> a >> b;
            D.merge(i, a, b);
        }else if (op == 2){ //回滚
            int k;
            cin >> k;
            rootfa[i] = rootfa[k];
            rootdep[i] = rootdep[k];
        }else if (op == 3){ //查询
            cin >> a >> b;
            rootfa[i] = rootfa[i - 1];
            rootdep[i] = rootdep[i - 1];
            a = D.find(i, a);
            b = D.find(i, b);
            if (a == b) cout << 1 << endl;
            else cout << 0 << endl;
        }
    }
}

int main(){
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int t = 1;
    //cin >> t;
    while(t--) solve();
    return 0;
}
```

#### 子序列自动机 ####

```cpp
#include <bits/stdc++.h>
#define endl '\n'
#define pll pair<i64, i64>
#define tll tuple<i64, i64, i64>
#define all(a) a.begin() + 1, a.end()
using namespace std;
using i64 = long long;
using db = long double;
const i64 N = 1e5 + 10;
const i64 mod = 998244353;
const i64 inf32 = 1e9;
const i64 inf64 = 5e18;

int typ, n, q, m;
int root[N];
struct PersistentTree{
    int ls[N << 5], rs[N << 5], nxt[N << 5], tot = 0;
    //单点修改
    inline void update(int & rt, int old, int l, int r, int p, int k){
        rt = ++tot;
        ls[rt] = ls[old], rs[rt] = rs[old], nxt[rt] = nxt[old];
        if (l == r){
            nxt[rt] = k;
            return;
        }
        int mid = l + r >> 1;
        if (p <= mid) update(ls[rt], ls[old], l, mid, p, k);
        else update(rs[rt], rs[old], mid + 1, r, p, k);
    }
    //单点查询
    inline int query(int rt, int l, int r, int p){
        if (l == r) return nxt[rt];
        int mid = l + r >> 1;
        if (p <= mid) return query(ls[rt], l, mid, p);
        else return query(rs[rt], mid + 1, r, p);
    } 
    //建树
    inline void build(vector<int> a){
        int n = a.size() - 1;
        for (int i = n; i >= 1; --i){
            update(root[i], root[i + 1], 1, m, a[i], i + 1);
        }
    }
}T;

void solve(){
    cin >> typ >> n >> q >> m;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; ++i) cin >> a[i];
    T.build(a);
    while (q--){
        int k;
        cin >> k;
        bool ok = true;
        int rt = 1;
        while (k--){
            int p;
            cin >> p;
            int to = T.query(root[rt], 1, m, p);
            if (!to) ok = false;
            if (to) rt = to;
        }
        if (ok) cout << "Yes" << endl;
        else cout << "No" << endl;
    }
}

int main(){
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int t = 1;
    //cin >> t;
    while(t--) solve();
    return 0;
}
```
